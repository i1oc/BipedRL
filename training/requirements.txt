# Python environment for training BipedRL with Unity ML-Agents.
#
# Requirements (from the official ML-Agents 4.2 installation docs):
#   - Python 3.10.1 - 3.10.12 (mlagents 1.1.0 refuses other versions).
#     The last Windows installer of the 3.10 series on python.org is 3.10.11.
#   - PyTorch 2.1.1 - 2.8.0. PyTorch 2.9+ breaks the ONNX export ML-Agents
#     uses, and mlagents 1.1.0 on PyPI does not cap it, so torch is pinned here.
#
# torch 2.8.0+cu128 is the newest allowed version and is built with CUDA 12.8,
# which RTX 50 series (Blackwell) GPUs need; it also runs on older NVIDIA GPUs.
# It needs an NVIDIA driver that supports CUDA 12.8 (version 570 or later).
#
# Setup (PowerShell, from the repository root):
#   py -3.10 -m venv .venv
#   .\.venv\Scripts\Activate.ps1
#   python -m pip install --upgrade pip
#   pip install -r training\requirements.txt

--extra-index-url https://download.pytorch.org/whl/cu128
torch==2.8.0+cu128
mlagents==1.1.0
